para comprobar si cuda quedó instalado

srun -p gpu-mig --exact -N 1 -n 1 -c 1 --time=00:05:00 \
  --gres=gpu:a100_1g.10gb:1 \
  "$VIRTUAL_ENV/bin/python" -c 'import torch; print("PyTorch:", torch.__version__); print("CUDA disponible:", torch.cuda.is_available()); print("GPU:", torch.cuda.get_device_name(0) if torch.cuda.is_available() else "ninguna")'





para probar MarioGPT

time srun -p gpu-mig --exact -N 1 -n 1 -c 1 --time=00:05:00 \
  --gres=gpu:a100_1g.10gb:1 \
